function snapshot(){

  var state={

    tables:JSON.parse(JSON.stringify(tables)),

    clientTables:JSON.parse(JSON.stringify(clientTables)),

    screens:JSON.parse(JSON.stringify(screens)),

  };

}

function chairSpacing(){

  // Calculate distance between chair centers in inches

  var rIn=cfg.tableDiam/2; // table radius in inches

  var cR=cfg.chairSizeIn/2; // chair radius in inches

  var ringR=rIn+cR+cfg.chairGapIn; // ring radius where chair centers sit (inches, +2in gap)

  var n=cfg.chairsPerTable;

  if(n<2)return 0;

  if(cfg.tableShape==="crescent"){

    // Chairs on ~82% of circle

    var arcSpan=Math.PI*0.82;

    var arcPerChair=arcSpan/n;

    return 2*ringR*Math.sin(arcPerChair/2);

  }else if(cfg.tableShape==="round"){

    // Full circle

    var arcPerChair2=2*Math.PI/n;

    return 2*ringR*Math.sin(arcPerChair2/2);

  }else{

    // Rectangular: approximate along long side

    var longSide=Math.max(cfg.rectTableW*12,cfg.rectTableD);

    var perSide=Math.floor(n/2);

    if(perSide<2)return longSide;

    return longSide/(perSide-1);

  }

}


function facingForWall(){

  // All tables face the same direction: chairs (+Y local) toward stage wall

  // rotate(0) = chairs point down (+Y screen). So:

  // stage top: chairs face up = rotate(180)

  // stage bottom: chairs face down = rotate(0)

  // stage left: chairs face left = rotate(90)  

  // stage right: chairs face right = rotate(270)

  var map={top:180,bottom:0,left:90,right:270};

  return map[cfg.stageWall]||0;

}

function faceToward(obj,target){

  // Per-object angle toward a specific point (used for individual rotate-toward)

  var dx=target.x-obj.x, dy=target.y-obj.y;

  if(Math.abs(dx)<0.01&&Math.abs(dy)<0.01)return facingForWall();

  var r = Math.atan2(-dx, dy) * 180 / Math.PI;

  return ((Math.round(r/15)*15) % 360 + 360) % 360;

}

function detectHotelWallsFromPdf(callback){

  if(!pdfBg || !pdfPageW || !pdfPageH){ if(callback) callback(false); return; }

  var img = new Image();

  img.onload = function(){

    var c = document.createElement("canvas");

    c.width = pdfPageW; c.height = pdfPageH;

    var ctx = c.getContext("2d");

    ctx.drawImage(img,0,0,pdfPageW,pdfPageH);

    var data = ctx.getImageData(0,0,pdfPageW,pdfPageH).data;


    function darkAt(x,y){

      var i=(y*pdfPageW+x)*4;

      var r=data[i], g=data[i+1], b=data[i+2], a=data[i+3];

      if(a<20) return 0;

      var lum=0.2126*r+0.7152*g+0.0722*b;

      return lum < 170 ? 1 : 0;

    }


    var rowScores=new Array(pdfPageH).fill(0);

    var colScores=new Array(pdfPageW).fill(0);

    for(var y=0;y<pdfPageH;y++){

      var rs=0;

      for(var x=0;x<pdfPageW;x++) rs += darkAt(x,y);

      rowScores[y]=rs;

    }

    for(var x=0;x<pdfPageW;x++){

      var cs=0;

      for(var y=0;y<pdfPageH;y++) cs += darkAt(x,y);

      colScores[x]=cs;

    }


    var rowThresh=Math.max(20, Math.max.apply(null,rowScores)*0.35);

    var colThresh=Math.max(20, Math.max.apply(null,colScores)*0.35);


    var yStart=Math.floor(pdfPageH*0.03), yEnd=Math.floor(pdfPageH*0.97);

    var xStart=Math.floor(pdfPageW*0.03), xEnd=Math.floor(pdfPageW*0.97);


    var top=yStart, bottom=yEnd, left=xStart, right=xEnd;

    for(var y=yStart;y<yEnd;y++){ if(rowScores[y]>=rowThresh){ top=y; break; } }

    for(var y=yEnd;y>=yStart;y--){ if(rowScores[y]>=rowThresh){ bottom=y; break; } }

    for(var x=xStart;x<xEnd;x++){ if(colScores[x]>=colThresh){ left=x; break; } }

    for(var x=xEnd;x>=xStart;x--){ if(colScores[x]>=colThresh){ right=x; break; } }


    if(right-left < pdfPageW*0.25 || bottom-top < pdfPageH*0.18){

      pdfWallBox = null;

      if(callback) callback(false);

      return;

    }


    pdfWallBox = {x:left, y:top, w:right-left, h:bottom-top};

    if(callback) callback(true);

  };

  img.onerror = function(){ if(callback) callback(false); };

  img.src = pdfBg;

}

function redistributeTables(){

  if(tables.length<1)return;snapshot();

  var m=met(),sr=sRect(),w=cfg.stageWall,sp=cfg.tableSpacing*PX;

  var n=tables.length;

  // Use manual cols/rows if set, otherwise auto square

  var cols,rows;

  if(cfg.tableCols>0&&cfg.tableRows>0){

    cols=cfg.tableCols;rows=cfg.tableRows;

    // If user set fewer slots than tables, expand rows

    while(cols*rows<n)rows++;

  }else if(cfg.tableCols>0){

    cols=cfg.tableCols;rows=Math.ceil(n/cols);

  }else if(cfg.tableRows>0){

    rows=cfg.tableRows;cols=Math.ceil(n/rows);

  }else{

    cols=Math.ceil(Math.sqrt(n));rows=Math.ceil(n/cols);

  }

  // Update config display values

  cfg.tableCols=cols;cfg.tableRows=rows;

  // Available area opposite the stage

  var aX,aY,aW,aH;

  if(w==="top"){aX=m.rX;aY=sr.y+sr.h+30;aW=m.rW;aH=m.rY+m.rH-sr.y-sr.h-40}

  else if(w==="bottom"){aX=m.rX;aY=m.rY+10;aW=m.rW;aH=sr.y-m.rY-40}

  else if(w==="left"){aX=sr.x+sr.w+30;aY=m.rY;aW=m.rX+m.rW-sr.x-sr.w-40;aH=m.rH}

  else{aX=m.rX+10;aY=m.rY;aW=sr.x-m.rX-40;aH=m.rH}

  var cellW=Math.max(sp,40);

  var cellH=Math.max(sp,40);

  var gridW=cellW*(cols-1);

  var gridH=cellH*(rows-1);

  var startX=aX+(aW-gridW)/2;

  var startY=aY+(aH-gridH)/2;

  var sf=sFront();

  for(var i=0;i<n;i++){

    var row=Math.floor(i/cols);

    var col=i%cols;

    // Count items in this row

    var rowStart=row*cols;

    var itemsThisRow=Math.min(cols,n-rowStart);

    // Center this row if it has fewer items

    var rowGridW=cellW*(itemsThisRow-1);

    var rowOffX=aX+(aW-rowGridW)/2;

    tables[i].x=rowOffX+col*cellW;

    tables[i].y=startY+row*cellH;

    if(tables[i].facing===undefined||tables[i].facing===null)tables[i].facing=facingForWall();

    tables[i].label=""+(i+1);

  }

  render();

}


function applyCustomLayout(){

  if(customLayout.length===0)return;

  snapshot();

  tables=[];

  var m=met(),sr=sRect(),w=cfg.stageWall,sp=cfg.tableSpacing*PX;

  // Available area

  var aX,aY,aW,aH;

  if(w==="top"){aX=m.rX;aY=sr.y+sr.h+30;aW=m.rW;aH=m.rY+m.rH-sr.y-sr.h-40}

  else if(w==="bottom"){aX=m.rX;aY=m.rY+10;aW=m.rW;aH=sr.y-m.rY-40}

  else if(w==="left"){aX=sr.x+sr.w+30;aY=m.rY;aW=m.rX+m.rW-sr.x-sr.w-40;aH=m.rH}

  else{aX=m.rX+10;aY=m.rY;aW=sr.x-m.rX-40;aH=m.rH}

  var totalRows=customLayout.length;

  var rowH=Math.max(sp,aH/(totalRows+1));

  var face=facingForWall();

  var num=1;

  for(var ri=0;ri<totalRows;ri++){

    var cols=customLayout[ri];

    if(cols<1)continue;

    var colW=Math.max(sp,aW/(cols+1));

    // Center this row

    var rowW=colW*(cols-1);

    var rowStartX=aX+(aW-rowW)/2;

    var rowY=aY+rowH*(ri+1);

    for(var ci2=0;ci2<cols;ci2++){

      tables.push({id:uid(),x:rowStartX+colW*ci2,y:rowY,label:""+num,facing:face});

      num++;

    }

  }

  // Update cfg to reflect

  cfg.tableRows=totalRows;

  cfg.tableCols=Math.max.apply(null,customLayout);

  render();

}


function parseCustomLayout(text){

  customLayoutText=text;

  customLayout=[];

  // Parse formats: "5,6,5" or "5-6-5" or "5 6 5" or multiline

  var parts=text.replace(/[\n\r]+/g,",").replace(/[\-\/|]+/g,",").split(",");

  for(var pi=0;pi<parts.length;pi++){

    var v=parseInt(parts[pi].trim());

    if(!isNaN(v)&&v>0)customLayout.push(v);

  }

  return customLayout;

}

  function drawWall(x1,y1,x2,y2,isHoriz){

    var segs=[{s:0,e:1}]; // normalized segments

    var len=isHoriz?(x2-x1):(y2-y1);

    for(var dj=0;dj<doors.length;dj++){

      var dr=doors[dj];

      // Check if door is near this wall

      var nearWall=false,doorPos=0;

      if(isHoriz){

        // Horizontal wall: check Y proximity

        var wy=y1;

        if(Math.abs(dr.y-wy)<15){nearWall=true;doorPos=(dr.x-(isHoriz?x1:y1))/len;}

      }else{

        // Vertical wall: check X proximity

        var wx=x1;

        if(Math.abs(dr.x-wx)<15){nearWall=true;doorPos=(dr.y-y1)/len;}

      }

      if(nearWall&&doorPos>-0.1&&doorPos<1.1){

        var gapHalf=dr.hw/len;

        var gapStart=doorPos-gapHalf;

        var gapEnd=doorPos+gapHalf;

        // Split segments around this gap

        var newSegs=[];

        for(var sk=0;sk<segs.length;sk++){

          var seg=segs[sk];

          if(gapEnd<=seg.s||gapStart>=seg.e){newSegs.push(seg);continue;}

          if(gapStart>seg.s)newSegs.push({s:seg.s,e:Math.max(seg.s,gapStart)});

          if(gapEnd<seg.e)newSegs.push({s:Math.min(seg.e,gapEnd),e:seg.e});

        }

        segs=newSegs;

      }

    }

    var wallH="";

    for(var sl2=0;sl2<segs.length;sl2++){

      var seg2=segs[sl2];

      if(seg2.e-seg2.s<0.001)continue;

      if(isHoriz){

        wallH+='<line x1="'+(x1+seg2.s*len)+'" y1="'+y1+'" x2="'+(x1+seg2.e*len)+'" y2="'+y2+'" stroke="#111" stroke-width="'+wallThick+'" stroke-linecap="square"/>';

      }else{

        wallH+='<line x1="'+x1+'" y1="'+(y1+seg2.s*len)+'" x2="'+x2+'" y2="'+(y1+seg2.e*len)+'" stroke="#111" stroke-width="'+wallThick+'" stroke-linecap="square"/>';

      }

    }

    return wallH;
